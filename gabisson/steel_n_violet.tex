\songtitle{Steel \& Violet}

\tag{thomas-entourage}\tag{gabriele-wiater}\tag{thomas-garcia}\tag{tallinn}

\begin{notes}
Based on the Thomas' Entourage universe and the volatile connection between Gabriele Wiater and Thomas Garcia in Tallinn. %
The song channels industrial menace and addictive chemistry through a hard-edged, rhythmic sound.
\end{notes}

\begin{style}
