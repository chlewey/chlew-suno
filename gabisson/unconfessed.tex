\songtitle{Unconfessed}

\tag*{2013}\tag{club-house}\tag{autobiographical}

\begin{notes}
A confession song about longing for Nohelia and the guilt of a feeling that can’t be shared. %
The lyrics explore a dangerous emotional edge, where desire persists despite responsibility and the weight of real life.
\end{notes}

\begin{style}
club house, dubstep drop, 143 BPM, distorted synth lead, wobble bass, sidechain pumping, filtered saw pads, metallic percussion, punchy kick, clipped snare, tape saturation, wide stereo mix, plate reverb, hard limiter, tense buildup, aggressive release, dark euphoria
\end{style}
