\songtitle{The Healer and the Hurting}

\tag{ricochets}
\begin{notes}
Based on charactes and scenes from \emph{Ricochets: When destinies overlap} by Carlos Thompson.\\
Character: Lidewijn Josefin André, \emph{Lida} André
\end{notes}
\begin{style}
colombian pasillo, narrative song, two male tenor vocals, two guitars, baritone lead, guitar tremolo, nylon-string arpeggios, bass counterline, brushed snare, accordion accents, minor key harmony, tape saturation, plate reverb, live room ambience, bittersweet longing, restrained pulse, rubato intro, three-quarter sway, 1960s folk recording
\end{style}
