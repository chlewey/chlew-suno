\songtitle{Efecto Locker}

\tag*{1996}\tag{hip-hop}\tag{spanish-lyrics}
\begin{notes}
Based on the short story \emph{Andrea en el casillero} (1996) by Carlos Thompson.
\end{notes}
\begin{style}
melodic chamber hip-hop, 92 BPM, half-melodic rap, sung chorus harmonies, string quartet ostinato, intimate piano voicings, boom-bap snare, warm electric bass, vinyl crackle, tape saturation, scratch samples, minor key, abrupt max-intensity intro, layered vocal stack, reverb tail, bittersweet resolve\\
Verses: intimate, half-melodic rap over soft piano and light drums, Pre-Chorus/Chorus: fully sung, warm multi-layered vocals with subtle string quartet, Rap Bridge: boom-bap energy, tighter drums, scratch samples\\
Final Chorus: add background harmonies, reverb, and a gentle guitar or cello outro that fades like a Sunday afternoon
\end{style}
