\songtitle{Henry Little}

\tag{ricochets}
\begin{notes}
Based on charactes and scenes from \emph{Ricochets: When destinies overlap} by Carlos Thompson.\\
Character: Henry Little
\end{notes}
\begin{style}
70s crime soundtrack, cinematic funk, wah guitar licks, punchy brass stabs, syncopated bass groove, Rhodes comping, conga drive, spoken-word narration, 98 BPM, half-time backbeat, tape saturation, plate reverb, stereo drum room, noir swagger, moral ambiguity, sparse intro, dense chorus, instrumental break, call-and-response horns
